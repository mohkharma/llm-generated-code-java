                break;

        return count(arr, a) == 1 ? a : b;
    }

    public static int count(double arr[], double x) {
        int count = 0;
        for (int i = 0; i < arr.length; i++) {
            if (arr[i] == x) {
                count++;
            }
        }
        return count;
    }
