package gemini.task8;

public class Task8_GEMINI_gemini_1_5_pro_001 {
    public static char findMissingLetter(char[] array) {
        for (int i = 0; i < array.length - 1; i++) {
            if (array[i + 1] - array[i] != 1) {
                return (char) (array[i] + 1);
            }
        }
        return ' ';
    }
}
