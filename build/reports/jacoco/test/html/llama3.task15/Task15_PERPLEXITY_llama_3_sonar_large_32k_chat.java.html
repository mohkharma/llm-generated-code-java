            tower[i] = String.format("%" + (nFloors - i - 1) + "s%" + (2 * i + 1) + "s%" + (nFloors - i - 1) + "s", "", String.valueOf(new char[2 * i + 1]).replace("\0", "*"), "");

        return tower;
    }
